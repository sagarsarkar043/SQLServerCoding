--1.Rank Customers by Total Revenue ( Window Function)
--Problem--Rank customers based on total purchase amount.

Select * from customers;
Select * from Orders;


Select Customers.CustomerID, SUM(Orders.Amount) as TotalRevenue , Rank () over(ORDER BY SUM(Orders.Amount) DESC) as Rank
from Customers join Orders on Customers.CustomerID=Orders.CustomerID
group by Customers.CustomerID;


--2--Running Total of Revenue (Window Function)
--Problem--Calculate cumulative revenue over time.


Select * from Sales;


Select Sales.SaleDate , SUM(Sales.Revenue) over(Order by Sales.SaleDate)as RunningTotal
from Sales;


--3-Second Highest Salary (Subquery)
-- Problem--Find second highest salary from Employees.

Select Max(Salary)  as SecondHighestSalary from Employees where Salary <(Select Max(Salary) from Employees);

--4️ Employees Earning Above Average (Subquery)
--Problem--Find employees earning more than average salary.

Select * from Employees where Salary > (Select AVG(Salary) from Employees);


--5--Remove Duplicate Orders (CTE + Window)
--Problem--Assume duplicates exist, remove them.

WITH CTE AS (
    SELECT *,
           ROW_NUMBER() OVER (PARTITION BY CustomerID, Amount ORDER BY OrderID) AS rn
    FROM Orders
)
DELETE FROM CTE WHERE rn > 1;


--6-LEFT JOIN (Customers without Orders)
-- Problem--Show all customers even if they have no orders.


Select Customers.CustomerID,Orders.OrderID from 
Customers left join Orders on
Customers.CustomerID= Orders.CustomerID
where OrderID is NULL;


--7--Self Join (Employee → Manager)
--Problem--Show employees and their managers.


Select e1.EmpName,e2.EmpName from Employees e1 left join Employees e2 
on e1.ManagerID=e2.EmpID;

--Problem-- Find top 2 highest orders per customer.
